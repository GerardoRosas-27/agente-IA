import asyncio
from typing import Optional

class QRCodeHandler:
    """
    Maneja el ciclo de vida del escaneo QR, desde la captura hasta el intento de conexión.
    Es responsable de correr las validaciones y los procesos asíncronos.
    """
    def __init__(self, connection_service):
        # Asumimos que 'connection_service' es un servicio inyectado para realizar el handshake.
        self.connection_service = connection_service

    async def process_scan(self, qr_data: str) -> tuple[bool, str]:
        """
        Procesa los datos del QR escaneados. Ejecuta validaciones y dispara la conexión si es válida.

        Args:
            qr_data: El string obtenido directamente del scanner/evento de escaneo.

        Returns:
            tuple[bool, str]: (success status, message)
        """
        if not qr_data or "ERROR" in qr_data.upper():
             return False, "Código QR no reconocido o datos nulos."

        try:
            # 1. Validación de Esquema
            validated_qr = self._validate_qr_format(qr_data)
            if not validated_qr:
                return False, "El código QR detectado tiene un formato inválido para este sistema."

            print(f"[QR Handler] Iniciando handshake con valor: {validated_qr}")
            # 2. Ejecución de Handshake asíncrono
            success, message = await self.connection_service.establish_connection(validated_qr)
            return success, message

        except TimeoutError:
            return False, "Tiempo de conexión agotado. Verifique la red o el servicio externo."
        except Exception as e:
            return False, f"Ocurrió un error desconocido durante el proceso: {str(e)}"


    @staticmethod
