from __future__ import annotations
import os
import requests
from typing import Optional, Any
import json


class WABusinessAPIClientError(Exception):
    """Base exception for WhatsApp Business API client errors."""
    pass


class APIAuthenticationError(WABusinessAPIClientError):
    """Raised when API credentials (Token/ID) are invalid or expired."""
    def __init__(self, message: str = "API Authentication Failed. Check your token and account ID."):
        super().__init__(message)

class RecipientNotFoundError(WABusinessAPIClientError):
    """Raised when the target phone number is invalid or blocked by WhatsApp API."""
    def __init__(self, message: str = "Recipient Error. The provided recipient number is invalid or inactive."):
        super().__init__(message)


class WABusinessApiClient:
    """
    Cliente dedicado para interactuar con la WhatsApp Business Cloud API de Meta.

    Encapsula la lógica de conexión, autenticación y envío de mensajes.
    """

    API_ENDPOINT = "https://graph.facebook.com/v18.0/" # Placeholder endpoint
    TOKEN_ENV = "WHATSAPP_ACCESS_TOKEN"
    ACCOUNT_ID_ENV = "WHATSAPP_BUSINESS_ACCOUNT_ID"


    def __init__(self):
        """Inicializa el cliente cargando credenciales de entorno."""
        self.access_token: Optional[str] = os.environ.get(self.TOKEN_ENV)
        self.account_id: Optional[str] = os.environ.get(self.ACCOUNT_ID_ENV)

        if not self.access_token or not self.account_id:
            raise ValueError("Missing required environment variables: WHATSAPP_ACCESS_TOKEN and WHATSAPP_BUSINESS_ACCOUNT_ID.")

        print("WABusinessApiClient initialized successfully.")


    def initialize_connection(self) -> bool:
        """
        Verifica y establece la conexión inicial con WhatsApp API.

        Simula la llamada de verificación al endpoint de Meta.
        Lanza APIAuthenticationError si las credenciales fallan.
        """
        print("Attempting to initialize connection...")
        headers = {
            "Authorization": f"Bearer {self.access_token}",
            "Content-Type": "application/json",
        }

        # Simulación de una llamada de verificación (Ej: obtener datos del perfil)
        try:
            # En un entorno real, harías un GET al endpoint /me o similar.
            response = requests.get(f"{self.API_ENDPOINT}{self.account_id}/me", headers=headers, timeout=10)

            if response.status_code != 200:
                # Manejo de errores comunes de API (400, 401, etc.)
                error_message = response.json().get("error", {}).get("message", "Unknown API error.")
                raise APIAuthenticationError(f"API connection failed (Status {response.status_code}): {error_message}")

            print("Connection established successfully.")
            return True

        except requests.exceptions.RequestException as e:
            # Captura errores de red o timeouts
            raise WABusinessAPIClientError(f"Network error during connection initialization: {e}")


    def send_test_message(self, to_number: str, message_text: str) -> bool:
        """
        Envía un mensaje de texto de prueba a un número destino.

        Args:
            to_number: El número del destinatario (ej: 'whatsapp:+521234567890').
            message_text: El contenido del mensaje.

        Returns:
            True si el mensaje fue encolado exitosamente.

        Raises:
            WABusinessAPIClientError: Si falla la API de WhatsApp (ej. número inválido).
        """
        print(f"Sending test message to {to_number}...")
        message_body = {
            "messaging_product": "whatsapp",
            "recipient": {"is_opted_in": True, "phone": to_number},
            "message": {"text": message_text}
        }

        headers = {
            "Authorization": f"Bearer {self.access_token}",
            "Content-Type": "application/json",
        }

        try:
            # Simulación del endpoint de envío de mensajes
            response = requests.post(f"{self.API_ENDPOINT}{self.account_id}/messages", headers=headers, json=message_body, timeout=10)

            if response.status_code == 200 and response.json().get("messaging_product") == "whatsapp":
                return True # Éxito al encolar el mensaje
            elif response.status_code == 400:
                # WhatsApp API suele devolver un error específico de negocio aquí
                error_data = response.json()
                if "invalid recipient" in str(error_data):
                    raise RecipientNotFoundError("WhatsApp rejected the number format or it is inactive.")
                else:
                     # General fallback para errores 400 que no son el número receptor
                    raise WABusinessAPIClientError(f"Bad Request from API (Code 400). Details: {error_data}")
            else:
                 # Otros errores de estado (5xx, etc.)
                status = response.status_code
                message = response.json().get("error", {}).get("message", "Unknown API status error.")
                raise WABusinessAPIClientError(f"Failed to send message. Status {status}. Details: {message}")

        except requests.exceptions.RequestException as e:
            # Captura errores de red o timeouts al enviar
            raise WABusinessAPIClientError(f"Network error during message sending: {e}")

from __future__ import annotations
import os
# Importamos el nuevo módulo y las excepciones
from skills.wa_business_api_client import WABusinessApiClient, APIAuthenticationError, WABusinessAPIClientError


class WhatsAppConnector:
    """
    Orquestador de funcionalidades relacionadas con WhatsApp Business API.
    Utiliza WABusinessApiClient para abstraer la complejidad HTTP.
    """

    def __init__(self):
        try:
            # Inicialización centralizada usando el nuevo cliente especializado
            self._client = WABusinessApiClient()
            self._is_connected = self._client.initialize_connection()
        except (ValueError, WABusinessAPIClientError) as e:
            print(f"Warning: WhatsApp Connector failed to initialize API connection: {e}")
            self._client = None
            self._is_connected = False

    def check_connectivity(self) -> bool:
        """Devuelve True si la conexión fue establecida correctamente."""
        return self._is_connected and self._client is not None


    def send_message(self, to_number: str, message_text: str) -> tuple[bool, str]:
        """
        Intenta enviar un mensaje de WhatsApp.

        Args:
            to_number: Número destino en formato E.164 con prefijo whatsapp:.
            message_text: Mensaje a enviar.

        Returns:
            Tuple (success_boolean, status_message).
        """
        if not self.check_connectivity():
            return False, "WhatsApp API is not connected. Check credentials or network."

        try:
            # Uso del nuevo cliente centralizado
            success = self._client.send_test_message(to_number, message_text)
            if success:
                return True, f"Message successfully enqueued to WhatsApp for {to_number}."
            else:
                return False, "Unknown failure during message enqueueing."

        except APIAuthenticationError as e:
            return False, f"AUTHENTICATION FAILURE: {e}"
        except WABusinessAPIClientError as e:
            # Captura errores de negocio (número inválido, etc.)
            return False, f"API BUSINESS LOGIC FAILURE: {e}"

    def process_incoming_webhook(self, payload: dict) -> None:
        """
        Procesa el JSON recibido por webhook. Debe ser llamado en un hilo/proceso aparte.
        """
        if self._client:
            self._client.process_webhook(payload)

## Riesgos / notas
1.  **Dependencia Externa:** La funcionalidad depende totalmente del uso correcto de las variables de entorno (`WHATSAPP_ACCESS_TOKEN`, `WHATSAPP_BUSINESS_ACCOUNT_ID`) y la disponibilidad de credenciales válidas en el entorno de ejecución real. Las pruebas unitarias asumen su presencia para simular fallos controladamente.
2.  **API Versión:** Se utiliza un endpoint simulado (`v18.0`). En caso de que Meta cambie sus endpoints o requerimientos, este módulo deberá ser actualizado por el desarrollador.
3.  **Webhook Asincronía:** El método `process_webhook` es un *placeholder*. La gestión real de webhooks (manejo del estado, deduplicación y enrutamiento) debe implementarse como un servicio dedicado fuera del ámbito de este cliente API simple.
