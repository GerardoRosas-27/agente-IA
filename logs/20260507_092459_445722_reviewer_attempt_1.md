```python:src/scanner_service.py
import asyncio
from typing import Callable, Awaitable, Dict, Any

class ScannerEvent:
    """Estructura para los eventos emitidos por el escáner."""
    def __init__(self, success: bool, payload: str = "", error: str = ""):
        self.success = success
        self.payload = payload
        self.error = error

class QRScannerListener:
    """
    Servicio desacoplado que escucha eventos de escaneo QR asíncronamente.
    Emite un evento a los suscriptores al detectar datos.
    """
    def __init__(self):
        # Mapa para almacenar las funciones callback (suscriptores)
        self._callbacks: list[Callable[[ScannerEvent], Awaitable[None]]] = []

    def subscribe(self, callback: Callable[[ScannerEvent], Awaitable[None]]):
        """Añade un callback a la lista de suscriptores."""
        self._callbacks.append(callback)

    async def _emit_event(self, event: ScannerEvent):
        """Emite el evento a todos los suscriptores simultáneamente."""
        await asyncio.gather(*[cb(event) for cb in self._callbacks])

    async def simulate_scan(self, success: bool, payload: str = "", error: str = ""):
        """Método público para simular la recepción de un escaneo (usado por el harness)."""
        event = ScannerEvent(success=success, payload=payload, error=error)
        print(f"[SCANNER] Evento recibido. Éxito: {success}")
        await self._emit_event(event)

    async def start_listening(self):
        """Simula la operación continua de escucha (en un sistema real usaría un asyncio.Queue)."""
        # En producción, esta tarea estaría atada a un puerto/dispositivo serial
        print("QRScannerListener iniciado y esperando eventos...")

```

**2. Modificación del State Manager (`src/state_manager.py`)**
Se asume la existencia de una clase `StateManager` existente. Se añaden métodos para manejar el nuevo estado y el timeout.

```python:src/state_manager.py
import asyncio
from enum import Enum

class SystemState(Enum):
    IDLE = "idle"
    WAITING_FOR_SCAN = "waiting_for_scan"
    CONNECTING = "connecting"
    FAILED = "failed"
    SUCCESS = "success"

class StateManager:
    """Gestiona el estado de la sesión, incluyendo transiciones asíncronas y timeouts."""
    def __init__(self):
        self._state: SystemState = SystemState.IDLE
        self.scan_timeout_task: asyncio.Task | None = None

    @property
    def state(self) -> SystemState:
        return self._state

    async def transition_to_waiting(self, qr_listener):
        """Inicia el estado de espera y gestiona el timeout."""
        if self.state != SystemState.IDLE and self.state != SystemState.FAILED:
            print("Advertencia: Transición a WAITING ignorada desde un estado no válido.")
            return

        self._state = SystemState.WAITING_FOR_SCAN
        print(f"[STATE] Transición exitosa a {SystemState.WAITING_FOR_SCAN.value}.")

        # 1. Limpiar tareas anteriores si existen
        if self.scan_timeout_task and not self.scan_timeout_task.done():
            self.scan_timeout_task.cancel()

        # 2. Crear la tarea de Timeout (ejemplo: 60 segundos)
        async def timeout_handler(delay: int, qr_listener):
            await asyncio.sleep(delay)
            if self._state == SystemState.WAITING_FOR_SCAN:
                print("\n[STATE] ERROR: Tiempo de espera QR excedido. Transición a FALLO.")
                self._state = SystemState.FAILED
                # Aquí se debería emitir un evento específico de timeout
        
        self.scan_timeout_task = asyncio.create_task(
            timeout_handler(60, qr_listener)
        )

    async def handle_qr_success(self):
        """Manejador llamado al recibir el QR."""
        if self._state != SystemState.WAITING_FOR_SCAN: return
        print("[STATE] Recepción exitosa de QR detectada.")
        # Cancela el timeout inmediatamente tras éxito
        if self.scan_timeout_task:
            self.scan_timeout_task.cancel()
        
        self._state = SystemState.SUCCESS

    async def handle_qr_failure(self):
        """Manejador llamado al recibir un QR inválido."""
        print("[STATE] Recepción de QR fallida o corrupta.")
        # El estado puede permanecer en WAITING_FOR_SCAN, pero se registra el fallo.
```
