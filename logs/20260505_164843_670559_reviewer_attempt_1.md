```python
# services/whatsapp_service.py
def generate_qr(number: str, message: str) -> str | None:
    """Genera el código QR en formato Base64 a partir de los parámetros WhatsApp."""
    # 1. Construir URL canónica (ej. wa.me?text=...)
    wa_url = build_whatsapp_link(number, message) 

    # 2. Generar imagen con qrcode y guardar en bytes
    img_data = generate_qrcode_image(wa_url) 

    # 3. Codificar a Base64 para transmisión fácil al frontend
    return base64.b64encode(img_data).decode('utf-8')
```
