import json
from typing import Any, Dict, Optional
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# --- Mocking / Utility Functions ---
# En un entorno real (FastAPI/Flask), estas funciones serían manejadas por el framework de routing.
def process_structured_event(event: Dict[str, Any]) -> None:
    """Simula la emisión del evento normalizado al resto del harness."""
    logger.info("✅ EVENTO ESTANDARIZADO EMITIDO:")
    print(json.dumps(event, indent=2))

def extract_message_data(message: Dict[str, Any]) -> Optional[Dict[str, Any]]:
    """
    Extrae y normaliza la información de un único mensaje de WhatsApp.
    :param message: Objeto 'message' del payload.
    :return: Diccionario normalizado o None si no es un mensaje válido.
    """
    data = {
        "sender_id": "N/A", # Este ID debería venir del contexto superior (entry)
        "timestamp": message.get("timestamp"),
        "event_type": None,
        "content": None,
        "media_detected": False,
    }

    if message.get("@type") == "text":
        data["event_type"] = "TEXT"
        data["content"] = message.get("text", "")
    elif message.get("@type") == "image":
        data["event_type"] = "MEDIA_IMAGE"
        media = message.get("image", {})
        if media and 'id' in media:
            data["media_detected"] = True
            # En un sistema real, aquí se usaría el ID para descargar binario
            data["media_ref_id"] = media['id']
    elif message.get("@type") == "video":
        data["event_type"] = "MEDIA_VIDEO"
        return None # Manejar otros tipos de medios según necesidad

    return data


# --- Webhook Handlers ---

def handle_webhook_challenge(mode: str, token: str, challenge: str) -> tuple[int, Dict[str, Any]]:
    """
    Maneja el handshake GET para verificar la propiedad del número.
    Retorna código de estado HTTP y cuerpo JSON.
    """
    # Se asume un token secreto preconfigurado en el ambiente o la configuración del sistema.
    SECRET_TOKEN = "MY_SUPER_SECRET_VERIFY_TOKEN" 

    if mode and token and challenge:
        if token == SECRET_TOKEN:
            logger.info("✅ Handshake de Webhook verificado con éxito.")
            return 200, {"hub.challenge": challenge}
        else:
            logger.warning("❌ Fallo en la verificación del Token.")
            return 403, {"error": "Unauthorized token"}

    logger.error("⚠️ Parámetros incompletos para el Handshake GET.")
    return 400, {"error": "Missing parameters"}


def handle_webhook_payload(payload: Dict[str, Any]) -> tuple[int, str]:
    """
    Procesa un payload POST de mensajes entrantes.
    Esta es la función principal del Listener.
    """
    if not payload or 'entry' not in payload:
        logger.warning("Payload recibido inválido o vacío.")
        return 200, "OK" # Siempre responder 200 para mantener el webhook activo

    try:
        for entry in payload.get('entry', []):
            # Asumimos que estamos interesados en la lista de cambios (changes)
            if 'changes' not in entry or entry['changes'][0].get('field') != 'messages':
                continue

            # Iterar sobre los eventos de mensaje dentro del cambio
            for change in entry['changes']:
                message = change.get('value', {}).get('message')
                if message:
                    sender_id = change['value'].get('metadata', {}).get('bca_user_id', 'UNKNOWN_SENDER')
                    
                    # 1. Normalizar los datos del mensaje
                    normalized_data = extract_message_data(message)
                    
                    if normalized_data:
                        # 2. Construir el evento estandarizado para el harness
                        standardized_event = {
                            "source": "whatsapp",
                            "event_type": "MESSAGE_RECEIVED",
                            "payload": {
                                "sender_id": sender_id,
                                "content": normalized_data['content'], 
                                "media_detected": normalized_data['media_detected'],
                                "media_ref_id": normalized_data.get('media_ref_id'),
                                "timestamp": normalized_data['timestamp']
                            }
                        }
                        # 3. Emisión del Evento
                        process_structured_event(standardized_event)

        return 200, f"Payload procesado con éxito: {len(payload['entry'])} entradas."

    except Exception as e:
        logger.error(f"Error fatal al procesar el payload de WhatsApp: {e}", exc_info=True)
        # Es crucial responder 200 incluso si falla el parsing interno, 
        # para no interrumpir la conexión con Meta/WhatsApp.
        return 200, f"Error interno en procesamiento: {str(e)}"

## Riesgos / notas
Se ha asumido que la estructura del *payload* JSON entrante es consistente con los mensajes recibidos vía Webhook (conteniendo `entry` -> `changes` -> `value` -> `message`). Si WhatsApp modifica significativamente esta estructura, el método `extract_message_data` deberá ser revisado. Además, la implementación simula la emisión del evento utilizando `print()` y se confía en que un *mock* de logging/event emitter en el módulo `process_structured_event` capturará este flujo para el resto del sistema.
