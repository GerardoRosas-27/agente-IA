        """
        Procesa los datos del QR escaneados. Ejecuta validaciones y dispara la conexión si es válida.

        Args:
            qr_data: El string obtenido directamente del scanner/evento de escaneo.

        Returns:
            tuple[bool, str]: (success status, message)
        """
        if not qr_data or "ERROR" in qr_data.upper():
             return False, "Código QR no reconocido o datos nulos."
