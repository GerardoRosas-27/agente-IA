```python:api_endpoints/whatsapp_hook.py
# ... (imports y setup anteriores)

def extract_message_data(entry, sender_id: str) -> dict | None:
    """
    Extrae y normaliza los datos de un mensaje entrante a un formato estandarizado.
    Asegura que la detección de contenido multimedia es precisa según el tipo de mensaje.
    """
    message = entry.get("message")
    if not message:
        return None

    event_data: dict[str, any] = {
        "sender_id": sender_id,
        "content": "",
        "media_detected": False, # Default value set to False
        "timestamp": entry.get("timestamp", None)
    }

    message_type = message.get("type")

    if message_type == "text":
        # Caso 1: Mensaje de Texto Puro
        event_data["content"] = str(message.get("text", ""))
        # media_detected se mantiene en False (Default)
        return event_data

    elif message_type == "image":
        # Caso 2: Contenido Multimedia (Imagen)
        image_id = message.get("image", {}).get("id")
        event_data["content"] = f"Image detected with ID: {image_id}"
        event_data["media_detected"] = True
        return event_data

    elif message_type == "document":
        # Caso 3: Contenido Multimedia (Documento)
        doc_id = message.get("document", {}).get("id")
        event_data["content"] = f"Document detected with ID: {doc_id}"
        event_data["media_detected"] = True
        return event_data

    # Manejar otros tipos de mensajes no cubiertos o nulos
    return None
```
